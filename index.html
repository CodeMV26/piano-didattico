/** * Funzione per generare il piano delle interrogazioni * @param {Array} studenti - Lista dei nomi degli studenti * @param {string} tipoAnno - "T" per Trimestri, "Q" per Quadrimestri * @param {Array} minInterrogazioniPerPeriodo - Numero minimo di interrogazioni per ciascun periodo (es. [2, 3] per quadrimestri o [2, 2, 2] per trimestri) * @param {number} studentiPerLezione - Quanti studenti interrogare per ciascuna lezione */ function generaPianoInterrogazioni(studenti, tipoAnno, minInterrogazioniPerPeriodo, studentiPerLezione = 2) { const periodi = []; // 1. Definizione della struttura in base a Trimestri (T) o Quadrimestri (Q) if (tipoAnno.toUpperCase() === 'T') { periodi.push( { nome: "1° Trimestre", min: minInterrogazioniPerPeriodo[0] || 2 }, { nome: "2° Trimestre", min: minInterrogazioniPerPeriodo[1] || 2 }, { nome: "3° Trimestre", min: minInterrogazioniPerPeriodo[2] || 2 } ); } else if (tipoAnno.toUpperCase() === 'Q') { periodi.push( { nome: "1° Quadrimestre", min: minInterrogazioniPerPeriodo[0] || 2 }, { nome: "2° Quadrimestre", min: minInterrogazioniPerPeriodo[1] || 2 } ); } else { throw new Error("Tipo anno non valido. Usare 'T' per Trimestri o 'Q' per Quadrimestri."); } const pianoDidattico = {}; // 2. Programmazione delle interrogazioni per ogni periodo periodi.forEach(periodo => { // Creiamo la lista con tutti gli studenti da interrogare in base al numero minimo del periodo let interrogazioniDaFare = []; studenti.forEach(studente => { for (let i = 0; i < periodo.min; i++) { interrogazioniDaFare.push(studente); } }); // Mescolamento casuale dell'ordine delle interrogazioni (Algoritmo Fisher-Yates) for (let i = interrogazioniDaFare.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [interrogazioniDaFare[i], interrogazioniDaFare[j]] = [interrogazioniDaFare[j], interrogazioniDaFare[i]]; } // Raggruppamento delle interrogazioni nelle singole lezioni const lezioni = []; for (let i = 0; i < interrogazioniDaFare.length; i += studentiPerLezione) { lezioni.push(interrogazioniDaFare.slice(i, i + studentiPerLezione)); } pianoDidattico[periodo.nome] = lezioni; }); return pianoDidattico; } // ========================================== // ESEMPIO DI UTILIZZO (Modifica con i tuoi dati) // ========================================== const elencoClasse = [ "Marco Rossi", "Giulia Bianchi", "Luca Verdi", "Sofia Russo", "Alessandro Esposito", "Emma Romano" ]; // Parametri di input: const tipoDivisione = "Q"; // "Q" per Quadrimestri, "T" per Trimestri const interrogazioniMinime = [2, 3]; // Es. 2 nel 1° quadrimestre, 3 nel 2° quadrimestre const interrogatiPerLezione = 2; // Generazione del piano const pianoGenerato = generaPianoInterrogazioni( elencoClasse, tipoDivisione, interrogazioniMinime, interrogatiPerLezione ); // Stampa del risultato console.log(JSON.stringify(pianoGenerato, null, 2));
