Scelta della suddivisione dell'anno let scelta = ""; while (scelta !== 'T' && scelta !== 'Q') { scelta = (await askQuestion("La tua scuola adotta Trimestri (T) o Quadrimestri (Q)? ")).trim().toUpperCase(); if (scelta !== 'T' && scelta !== 'Q') { console.log("Scelta non valida. Inserisci 'T' per trimestri o 'Q' per quadrimestri."); } } let periodi = []; if (scelta === 'T') { periodi = [ { nome: "1° Trimestre", minInterrogazioni: 0 }, { nome: "2° Trimestre", minInterrogazioni: 0 }, { nome: "3° Trimestre", minInterrogazioni: 0 } ]; const minInt = parseInt(inputVal, 10); if (!isNaN(minInt) && minInt > 0) { p.minInterrogazioni = minInt; break; } console.log("Inserisci un numero valido maggiore di 0."); } } return periodi; } function programmoInterrogazioni(studenti, periodi, interrogazioniPerLezione) { const pianoDidattico = {}; console.log("\n========================================"); console.log(" PROGRAMMAZIONE DELLE INTERROGAZIONI"); console.log("========================================"); for (let p of periodi) { // Generiamo la lista di tutte le interrogazioni necessarie per il periodo let interrogazioniDaFare = []; for (let studente of studenti) { for (let i = 0; i < p.minInterrogazioni; i++) { interrogazioniDaFare.push(studente); } } // Mescoliamo casualmente la lista (Algoritmo Fisher-Yates) for (let i = interrogazioniDaFare.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [interrogazioniDaFare[i], interrogazioniDaFare[j]] = [interrogazioniDaFare[j], interrogazioniDaFare[i]]; } // Raggruppiamo le interrogazioni per lezione const calendarioPeriodo = []; } return pianoDidattico; } function stampaProgrammazione(pianoDidattico) { for (let [periodo, lezioni] of Object.entries(pianoDidattico)) { console.log(`\n--- ${periodo.toUpperCase()} ---`); lezioni.forEach((studenti, index) => { console.log(` Lezione ${index + 1}: ${studenti.join(", ")}`); }); } } // ESECUZIONE async function main() { const classe = ["Marco Rossi", "Giulia Bianchi", "Luca Verdi", "Sofia Russo", "Alessandro Esposito", "Emma Romano"]; const periodiScolastici = await acquisisciImpostazioni();
